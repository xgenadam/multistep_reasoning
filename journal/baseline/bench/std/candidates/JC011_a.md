class Solution:
    def countWays(self, nums: List[int]) -> int:
        nums.sort()
        n = len(nums)
        count = 0
        
        # Check for selecting no students
        if nums[0] > 0:
            count += 1
        
        # Check for selecting k students where 1 <= k <= n
        for k in range(1, n + 1):
            if nums[k - 1] < k and (k == n or nums[k] >= k):
                count += 1
        
        return count
