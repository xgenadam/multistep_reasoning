from typing import List

class Solution:
    def minOperations(self, nums: List[int], k: int) -> int:
        current_xor = 0
        for num in nums:
            current_xor ^= num
        
        if current_xor == k:
            return 0
        
        target_xor = current_xor ^ k
        min_operations = 0
        
        for num in nums:
            if target_xor == 0:
                break
            num_xor = num ^ target_xor
            if num_xor < num:
                min_operations += 1
                target_xor ^= num ^ num_xor
        
        return min_operations
